CREATE TABLE USERS (
    USER_ID     INT IDENTITY(1,1) PRIMARY KEY,
    FIRST_NAME  NVARCHAR(MAX),
    LAST_NAME   NVARCHAR(MAX),
    STD_NUM     INT,
    CYS         NVARCHAR(MAX),
    SEX         NVARCHAR(MAX),
    USERNAME    NVARCHAR(MAX),
    EMAIL       NVARCHAR(MAX),
    PASSWORD    NVARCHAR(MAX) 
);

CREATE TABLE ADMIN_LOGIN (
    ADMIN_NUMBER INT PRIMARY KEY,
    USERNAME     NVARCHAR(MAX),
    PASSWORD     NVARCHAR(MAX)
);

CREATE TABLE LISTINGS (
    LISTING_ID      INT IDENTITY(1,1) PRIMARY KEY,
    USER_ID         INT NOT NULL FOREIGN KEY REFERENCES USERS(USER_ID),
    TITLE           NVARCHAR(100) NOT NULL,
    DESCRIPTION     NVARCHAR(MAX),
    PRICE           DECIMAL(10,2) NOT NULL,
    CATEGORY        NVARCHAR(50) NOT NULL,
    CONDITION       NVARCHAR(20) NOT NULL,
    STATUS          NVARCHAR(20) NOT NULL DEFAULT 'Available',
    MEETUP_SPOT     NVARCHAR(100),
    PAYMENT_METHOD  NVARCHAR(100),
    DATE_POSTED     DATETIME NOT NULL DEFAULT GETDATE()
);

CREATE TABLE LISTING_IMG (
    IMG_ID      INT IDENTITY(1,1) PRIMARY KEY,
    LISTING_ID  INT NOT NULL FOREIGN KEY REFERENCES LISTINGS(LISTING_ID),
    FILE_PATH   NVARCHAR(500) NOT NULL,
    IS_PRIMARY  BIT NOT NULL DEFAULT 0
);
